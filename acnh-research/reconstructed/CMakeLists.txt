cmake_minimum_required(VERSION 3.24)
project(acnh_reconstructed LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(acnh_modules src/bcsv.cpp src/pilot_flow.cpp src/constant_zero.cpp src/identifier_hash.cpp src/flag_int.cpp src/npad_full_key.cpp src/access_key.cpp src/safe_string_token.cpp)
target_include_directories(acnh_modules PUBLIC include)
add_library(identifier_hash SHARED src/identifier_hash.cpp)
target_include_directories(identifier_hash PUBLIC include)
set_target_properties(identifier_hash PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
add_library(flag_int SHARED src/flag_int.cpp)
target_include_directories(flag_int PUBLIC include)
set_target_properties(flag_int PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
add_library(npad_full_key SHARED src/npad_full_key.cpp)
target_include_directories(npad_full_key PUBLIC include)
set_target_properties(npad_full_key PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
add_library(access_key SHARED src/access_key.cpp)
target_include_directories(access_key PUBLIC include)
set_target_properties(access_key PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
add_library(safe_string_token SHARED src/safe_string_token.cpp)
target_include_directories(safe_string_token PUBLIC include)
set_target_properties(safe_string_token PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
if(MSVC)
  target_compile_options(acnh_modules PRIVATE /W4 /WX)
else()
  target_compile_options(acnh_modules PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(module_tests tests/module_tests.cpp)
target_link_libraries(module_tests PRIVATE acnh_modules)
add_executable(bcsv_probe tests/bcsv_probe.cpp)
target_link_libraries(bcsv_probe PRIVATE acnh_modules)
enable_testing()
add_test(NAME reconstructed_contracts COMMAND module_tests)
